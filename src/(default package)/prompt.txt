import io.restassured.RestAssured;
import io.restassured.path.json.JsonPath;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;


 * Class Name: Basics
 *
 * Purpose:
 * This class validates the Add Place API using Rest Assured.
 *
 * API Scenario:
 * 1. Set base URI
 * 2. Prepare request using query parameter, header, and body
 * 3. Hit POST API
 * 4. Validate status code, response body, and response header
 * 5. Extract full response as String
 * 6. Parse JSON response using JsonPath
 * 7. Extract place_id from response
 */
public class Basics {

    public static void main(String[] args) {

        // Base URI means main domain/root URL of the API.
        // Memory Hint: baseURI = API oda main address.
        RestAssured.baseURI = "https://rahulshettyacademy.com";

        // Payload means request body.
        // Here we are getting JSON body from Payload class using static method.
        // This keeps main test code clean.
        String requestPayload = Payload.addPlacePayload();

        // Sending POST request and storing complete response as String.
        String addPlaceResponse =
                given()
                        // Logs full request details: URL, headers, query params, body.
                        // Useful for debugging when API fails.
                        .log().all()

                        // Query parameter is added after ? in URL.
                        // Final URL will contain: ?key=qaclick123
                        .queryParam("key", "qaclick123")

                        // Header tells server that we are sending JSON body.
                        .header("Content-Type", "application/json")

                        // body() sends JSON payload/request body to API.
                        .body(requestPayload)

                .when()
                        // post() hits the Add Place API endpoint/resource.
                        .post("/maps/api/place/add/json")

                .then()
                        // assertThat() is used to validate response.
                        .assertThat()

                        // Validates HTTP status code.
                        // 200 means request is successful.
                        .statusCode(200)

                        // Validates response body field "scope" value is "APP".
                        .body("scope", equalTo("APP"))

                        // Validates response header.
                        // Note: Header values like Server can change, so this can be brittle.
                        .headers("Server", "Apache/2.4.52 (Ubuntu)")

                        // extract() allows us to take response data for later use.
                        .extract()

                        // response() gets the complete API response.
                        .response()

                        // Converts response object into String format.
                        .asString();

        // Prints complete Add Place API response.
        System.out.println(addPlaceResponse);

        // JsonPath is used to parse JSON response.
        // Memory Hint: JsonPath = JSON response la irunthu value edukka use pannuvom.
        JsonPath jsonPath = new JsonPath(addPlaceResponse);

        // Extracting place_id from response JSON.
        // This place_id can be reused in Get Place, Update Place, Delete Place APIs.
        String placeId = jsonPath.getString("place_id");

        // Prints extracted place_id.
        System.out.println("Place ID: " + placeId);
    }
}
